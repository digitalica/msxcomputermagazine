void main(int argv, char * argc[])
{
union _REGS inregs,outregs;

    if(argv > 1) {strncpy(flopname,argc[1],sizeof(flopname));FloppyParm=1;}

    fm_init();

#ifdef protected
    goprot();
#endif

    read_conf();


    sprite_init();
    if (!mem_init()
     || !video_init())
    {
	printf("Problems allocating memory!");
	exit(0);
    }
    read_roms();
    keyb_init();
    Disk_init();
    loadsram();
    z80_reset();
    io_init();
    snd_reset();
    snd_init();
    for(;;)
    {
	z80_exec();
	msx_dbug();
	if(halt && !imask) break;
    }
    savesram();
    save_conf();
    sprite_exit();
    video_exit();
    keyb_exit();
    Disk_exit();
    io_exit();
    snd_off();
    inregs.x.ax=_TEXTC80; _int86(0x10,&inregs,&outregs);
    exit(0);
}
